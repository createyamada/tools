import datetime
import math
import numbers
import os
import re
import sys
from pathlib import Path
from tkinter import Tk, filedialog

import openpyxl


MAX_SEARCH_DEPTH = 2
OUTPUT_DIR_NAME = "output_sql"


def select_input_directory():
    """フォルダ選択ダイアログを表示し、選択されたパスを返す。"""
    root = Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    try:
        selected = filedialog.askdirectory(title="変換対象のxlsxがあるフォルダを選択してください")
    finally:
        root.destroy()

    return Path(selected) if selected else None


def find_xlsx_files(input_dir):
    """選択フォルダ、子フォルダ、孫フォルダにあるxlsxを列挙する。"""
    files = []

    for current_dir, dir_names, file_names in os.walk(input_dir):
        current_path = Path(current_dir)
        depth = len(current_path.relative_to(input_dir).parts)

        # 出力先と、孫より深いディレクトリは検索しない。
        dir_names[:] = [
            name
            for name in dir_names
            if name != OUTPUT_DIR_NAME and depth < MAX_SEARCH_DEPTH
        ]

        for file_name in file_names:
            path = current_path / file_name
            if path.suffix.lower() == ".xlsx" and not file_name.startswith("~$"):
                files.append(path)

    return sorted(files, key=lambda path: str(path).lower())


def get_table_name(ws):
    origin_all_text = ws["B4"].value
    if origin_all_text is None:
        return None

    match = re.search(r"「(.*?)[)）]?」", str(origin_all_text))
    return match.group(1).strip() if match else None


def make_output_path(output_dir, table_name, input_file, sheet_name):
    """同名SQLが存在する場合も上書きしない出力パスを作る。"""
    safe_table_name = re.sub(r'[\\/:*?"<>|]', "_", table_name)
    safe_input_name = re.sub(r'[\\/:*?"<>|]', "_", input_file.stem)
    safe_sheet_name = re.sub(r'[\\/:*?"<>|]', "_", sheet_name)

    output_file = output_dir / f"{safe_table_name}.sql"
    if not output_file.exists():
        return output_file

    output_file = output_dir / (
        f"{safe_table_name}_{safe_input_name}_{safe_sheet_name}.sql"
    )
    suffix = 2
    while output_file.exists():
        output_file = output_dir / (
            f"{safe_table_name}_{safe_input_name}_{safe_sheet_name}_{suffix}.sql"
        )
        suffix += 1
    return output_file


def quote_identifier(value):
    """テーブル名・列名を標準SQLの識別子として引用する。"""
    return f'"{str(value).replace(chr(34), chr(34) * 2)}"'


def to_sql_literal(value):
    """Excelセルの値をSQLの値表現へ変換する。"""
    if value is None:
        return "NULL"
    if isinstance(value, bool):
        return "TRUE" if value else "FALSE"
    if isinstance(value, datetime.datetime):
        return f"'{value.isoformat(sep=' ')}'"
    if isinstance(value, (datetime.date, datetime.time)):
        return f"'{value.isoformat()}'"
    if isinstance(value, numbers.Number):
        if isinstance(value, float) and not math.isfinite(value):
            return "NULL"
        return str(value)

    escaped = str(value).replace("'", "''")
    return f"'{escaped}'"


def make_insert_statement(table_name, headers, record):
    columns = ", ".join(quote_identifier(header) for header in headers)
    values = ", ".join(to_sql_literal(record.get(header)) for header in headers)
    return f"INSERT INTO {quote_identifier(table_name)} ({columns}) VALUES ({values});"


def convert_workbook(input_file, output_dir):
    """1つのxlsxを開き、各シートのレコードをINSERT文へ変換する。"""
    print(f"\nファイル：'{input_file}'の処理を開始します。")
    wb = openpyxl.load_workbook(input_file, data_only=True, read_only=True)
    converted_count = 0

    try:
        for sheet_name in wb.sheetnames:
            ws = wb[sheet_name]
            table_name = get_table_name(ws)
            if not table_name:
                print(f"caution:{sheet_name}:B4セルからテーブル名を取得できません。スキップします。")
                continue

            base_header = ws["C7"].value
            if base_header is None:
                print(f"caution:{sheet_name}:C7セルに値がありません。スキップします。")
                continue
            base_header = str(base_header).strip()

            print(f"シート：{sheet_name}を処理中...(基準キー：{base_header})")
            rows = list(ws.iter_rows(values_only=True))
            records = []
            headers = []

            i = 0
            while i < len(rows):
                row = rows[i]
                if any(str(cell).strip() == base_header for cell in row if cell is not None):
                    block = []
                    j = i
                    while j < len(rows) and any(cell is not None for cell in rows[j]):
                        values = list(rows[j])
                        if len(values) > 1:
                            values[1] = None
                        block.append(tuple(values))
                        j += 1

                    keys = []
                    key_rows = []
                    for block_row in block:
                        first_val = next(
                            (str(cell).strip() for cell in block_row if cell is not None),
                            None,
                        )
                        if first_val:
                            keys.append(first_val)
                            key_rows.append(block_row)

                    if not headers:
                        headers = keys

                    max_len = max((len(block_row) for block_row in key_rows), default=0)
                    for col in range(1, max_len):
                        record = {}
                        valid = False
                        for key, block_row in zip(keys, key_rows):
                            value = block_row[col] if col < len(block_row) else None
                            record[key] = value
                            if value is not None:
                                valid = True

                        if valid and record.get(base_header) != base_header:
                            records.append(record)
                    i = j
                else:
                    i += 1

            if not records:
                print(f"caution:{sheet_name}:レコードが抽出されませんでした。")
                continue

            output_file = make_output_path(output_dir, table_name, input_file, sheet_name)
            with output_file.open("w", newline="\n", encoding="utf-8") as sql_file:
                sql_file.write(f"-- Source: {input_file.name} / {sheet_name}\n")
                for record in records:
                    statement = make_insert_statement(table_name, headers, record)
                    sql_file.write(f"{statement}\n")

            converted_count += 1
            print(f"success:{len(records)}件のレコードを'{output_file}'に出力しました。")
    finally:
        wb.close()

    return converted_count


def main():
    input_dir = select_input_directory()
    if input_dir is None:
        print("フォルダが選択されなかったため、処理を終了します。")
        return 0

    xlsx_files = find_xlsx_files(input_dir)
    if not xlsx_files:
        print("選択フォルダ、子フォルダ、孫フォルダにxlsxファイルがありません。")
        return 0

    output_dir = input_dir / OUTPUT_DIR_NAME
    output_dir.mkdir(exist_ok=True)
    print(f"{len(xlsx_files)}件のxlsxファイルが見つかりました。")

    success_files = 0
    failed_files = 0
    converted_sheets = 0
    for index, input_file in enumerate(xlsx_files, start=1):
        print(f"\n[{index}/{len(xlsx_files)}] {input_file}")
        try:
            converted_sheets += convert_workbook(input_file, output_dir)
            success_files += 1
        except Exception as error:
            failed_files += 1
            print(f"error:'{input_file}'の変換に失敗しました：{error}")

    print(
        f"\n処理完了：成功 {success_files}ファイル、失敗 {failed_files}ファイル、"
        f"SQL出力 {converted_sheets}件（出力先：'{output_dir}'）"
    )
    return 1 if failed_files else 0


if __name__ == "__main__":
    sys.exit(main())
